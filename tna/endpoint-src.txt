 * HOW TO USE THIS FILE

 * You already have a web app writing the OND tabs. Add this file to that same

 * actions into your existing doPost / doGet routers:
 *
 *   function doPost(e) {

 *     var hit  = NR_route(body);          // <- add this line
 *     if (hit) return hit;                // <- and this one
 *     ... your existing OND routing ...
 *   }
 *
 *   function doGet(e) {
 *     var hit = NR_routeGet(e.parameter); // <- add this line
 *     if (hit) return hit;                // <- and this one
 *     ... your existing OND routing ...
 *   }
 *
 * Every action name below is prefixed "nr." so it can never collide with your
 * OND actions. If NR_route sees an action without that prefix it returns null
 * and your existing code handles the request untouched.
 *
 * After adding the file: Deploy > Manage deployments > edit the existing
 * deployment > Version: New version > Deploy. The web app URL does not change.
 *
 * Run NR_setup() once from the editor to create the five tabs with their
 * headers. It is safe to run again; it never touches existing rows.

/* ------------------------------------------------------------------ utils */

function NR_ss_() {
  return SpreadsheetApp.openById(NR_SS_ID);
}

function NR_now_() {
  return Utilities.formatDate(new Date(), NR_TZ, 'dd MMM yyyy, HH:mm');
}

function NR_json_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function NR_tab_(key) {
  var ss = NR_ss_();
  var name = NR_TABS[key];
  var sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    sh.appendRow(NR_HEADERS[key]);
    sh.setFrozenRows(1);
    sh.getRange(1, 1, 1, NR_HEADERS[key].length).setFontWeight('bold');
  }
  return sh;
}

/** Creates all five tabs with headers. Safe to run more than once. */
function NR_setup() {
  Object.keys(NR_TABS).forEach(function (k) { NR_tab_(k); });
  return 'Nov Refresher tabs ready.';
}

function NR_str_(v) {
  return v === null || v === undefined ? '' : String(v);
}

function NR_num_(v) {
  var n = Number(v);
  return isNaN(n) ? '' : n;
}

/** Short, unambiguous code: no 0/O/1/I/5/S. */
function NR_code_() {
  var alphabet = 'ABCDEFGHJKLMNPQRTUVWXYZ2346789';
  var out = '';
  for (var i = 0; i < 5; i++) {
    out += alphabet.charAt(Math.floor(Math.random() * alphabet.length));
  }
  return out;
}

function NR_key_(parts) {
  var raw = parts.join('|').toLowerCase();
  var bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, raw);
  return Utilities.base64EncodeWebSafe(bytes).substring(0, 16);
}

/** Row index of the first row whose column `col` equals `value`, else -1. */
function NR_findRow_(sh, col, value) {
  var last = sh.getLastRow();
  if (last < 2) return -1;
  var vals = sh.getRange(2, col, last - 1, 1).getValues();
  for (var i = 0; i < vals.length; i++) {
    if (String(vals[i][0]).trim() === String(value).trim()) return i + 2;
  }
  return -1;
}

/* ------------------------------------------------------------- code issue */

/**
 * nr.issueCode — trainer opens a session and gets a code counsellors type in.
 * Reissuing for the same session returns the existing code instead of a new one,
 * so a trainer who reloads the panel does not invalidate codes already read out.
 */
function NR_issueCode_(p) {
  var sh = NR_tab_('codes');
  var session = NR_str_(p.session);
  var row = NR_findRow_(sh, 1, session);

  if (row > 0 && !p.rotate) {
    var existing = sh.getRange(row, 1, 1, NR_HEADERS.codes.length).getValues()[0];
    return NR_json_({ ok: true, code: existing[7], session: session, reused: true });
  }

  var code = NR_code_();
  var openUntil = NR_str_(p.openUntil);
  var values = [
    session, NR_str_(p.date), NR_str_(p.day), NR_str_(p.time), NR_str_(p.team),
    NR_str_(p.trainer), NR_str_(p.topics), code, openUntil,
    NR_str_(p.pin), NR_now_()
  ];

  if (row > 0) {
    sh.getRange(row, 1, 1, values.length).setValues([values]);
  } else {
    sh.appendRow(values);
  }
  return NR_json_({ ok: true, code: code, session: session, reused: false });
}

/** Looks a code up and reports whether it is still open. */
function NR_checkCode_(code, session) {
  var sh = NR_tab_('codes');
  var last = sh.getLastRow();
  if (last < 2) return { ok: false, reason: 'no_codes' };

  var rows = sh.getRange(2, 1, last - 1, NR_HEADERS.codes.length).getValues();
  var want = String(code || '').trim().toUpperCase();

  for (var i = 0; i < rows.length; i++) {
    if (String(rows[i][7]).trim().toUpperCase() !== want) continue;
    if (session && String(rows[i][0]).trim() !== String(session).trim()) {
      return { ok: false, reason: 'wrong_session', session: rows[i][0] };
    }
    var until = rows[i][8];
    if (until) {
      var closes = new Date(until);
      if (!isNaN(closes.getTime()) && new Date() > closes) {
        return { ok: false, reason: 'closed' };
      }
    }
    return {
      ok: true,
      session: rows[i][0], date: rows[i][1], day: rows[i][2], time: rows[i][3],
      team: rows[i][4], trainer: rows[i][5], topics: rows[i][6]
    };
  }
  return { ok: false, reason: 'not_found' };
}

/* -------------------------------------------------------------- 1. attend */

function NR_attendance_(p) {
  var check = NR_checkCode_(p.code, p.session);
  if (!check.ok) return NR_json_({ ok: false, error: check.reason });

  var sh = NR_tab_('attendance');
  var email = NR_str_(p.email).trim().toLowerCase();
  var key = NR_key_([check.session, email]);

  /* One check-in per person per session. A second submit updates the time
     rather than adding a duplicate row. */
  var row = NR_findRow_(sh, 11, key);
  var values = [
    NR_now_(), check.session, check.date, check.time, check.team,
    check.trainer, check.topics, NR_str_(p.name).trim(), email,
    String(p.code).trim().toUpperCase(), key
  ];

  if (row > 0) {
    sh.getRange(row, 1, 1, values.length).setValues([values]);
    return NR_json_({ ok: true, duplicate: true, session: check.session });
  }
  sh.appendRow(values);
  return NR_json_({ ok: true, duplicate: false, session: check.session });
}

/* ------------------------------------------------------------- 2. feedback */

function NR_feedback_(p) {
  var sh = NR_tab_('feedback');
  var email = NR_str_(p.email).trim().toLowerCase();
  var session = NR_str_(p.session);
  var key = NR_key_([session, email, 'fb']);

  var row = NR_findRow_(sh, 15, key);
  var values = [
    NR_now_(), session, NR_str_(p.date), NR_str_(p.time), NR_str_(p.team),
    NR_str_(p.trainer), NR_str_(p.topics),
    NR_num_(p.overall), NR_num_(p.clarity), NR_num_(p.usefulness),
    NR_str_(p.worked), NR_str_(p.better), NR_str_(p.name).trim(), email, key
  ];

  if (row > 0) {
    sh.getRange(row, 1, 1, values.length).setValues([values]);
    return NR_json_({ ok: true, updated: true });
  }
  sh.appendRow(values);
  return NR_json_({ ok: true, updated: false });
}

/* ---------------------------------------------------------------- 3. pulse */

/**
 * Anonymous. Nothing identifying is written: the page sends no name, no email
 * and no device key for a pulse, and this function stores none even if one
 * arrives, so a row cannot be traced to a counsellor.
 *
 * The pulse form's questions are not hardcoded. The page sends
 *   answers: [{question: "...", answer: "..."}, ...]
 * and this widens the header row the first time it meets a new question, so you
 * can change the pulse form without editing this script.
 */
function NR_pulse_(p) {
  var sh = NR_tab_('pulse');
  var answers = p.answers || [];
  var header = sh.getRange(1, 1, 1, Math.max(sh.getLastColumn(), NR_HEADERS.pulse.length)).getValues()[0];

  /* Trim trailing blanks so new questions append cleanly. */
  while (header.length && NR_str_(header[header.length - 1]) === '') header.pop();

  answers.forEach(function (a) {
    var q = NR_str_(a.question).trim();
    if (q && header.indexOf(q) === -1) header.push(q);
  });
  sh.getRange(1, 1, 1, header.length).setValues([header]);
  sh.getRange(1, 1, 1, header.length).setFontWeight('bold');
  sh.setFrozenRows(1);

  var rid = NR_str_(p.responseId).trim() || ('R' + new Date().getTime().toString(36).toUpperCase());

  var row = new Array(header.length).fill('');
  row[0] = NR_now_();
  row[1] = NR_str_(p.session);
  row[2] = NR_str_(p.date);
  row[3] = NR_str_(p.team);
  row[4] = NR_str_(p.scope);
  row[5] = rid;

  answers.forEach(function (a) {
    var at = header.indexOf(NR_str_(a.question).trim());
    if (at > -1) row[at] = NR_str_(a.answer);
  });

  var existing = NR_findRow_(sh, 6, rid);
  if (existing > 0) {
    sh.getRange(existing, 1, 1, row.length).setValues([row]);
    return NR_json_({ ok: true, updated: true });
  }
  sh.appendRow(row);
  return NR_json_({ ok: true, updated: false });
}

/* ------------------------------------------------------------ 4. ask ahead */

function NR_ask_(p) {
  var sh = NR_tab_('ask');
  var asker = NR_key_([NR_str_(p.deviceKey), NR_str_(p.email)]);
  var qid = 'Q' + new Date().getTime().toString(36).toUpperCase();

  sh.appendRow([
    NR_now_(), NR_str_(p.session), NR_str_(p.date), NR_str_(p.time), NR_str_(p.team),
    NR_str_(p.trainer), NR_str_(p.topics), NR_str_(p.scope),
    NR_str_(p.question).trim(), NR_str_(p.name).trim(), 0, 'No', qid, asker, ''
  ]);
  return NR_json_({ ok: true, questionId: qid });
}

/** One vote per device. Voting twice removes the vote. */
function NR_upvote_(p) {
  var sh = NR_tab_('ask');
  var row = NR_findRow_(sh, 13, NR_str_(p.questionId));
  if (row < 0) return NR_json_({ ok: false, error: 'not_found' });

  var voter = NR_key_([NR_str_(p.deviceKey)]);
  var cell = sh.getRange(row, 15);
  var voters = NR_str_(cell.getValue()).split(',').filter(function (v) { return v; });

  var at = voters.indexOf(voter);
  if (at > -1) { voters.splice(at, 1); } else { voters.push(voter); }

  cell.setValue(voters.join(','));
  sh.getRange(row, 11).setValue(voters.length);
  return NR_json_({ ok: true, count: voters.length, voted: at === -1 });
}

/** Questions for one session, or for the whole day when `day` is given. */
function NR_questions_(params) {
  var sh = NR_tab_('ask');
  var last = sh.getLastRow();
  if (last < 2) return NR_json_({ ok: true, questions: [] });

  var rows = sh.getRange(2, 1, last - 1, NR_HEADERS.ask.length).getValues();
  var wantSession = NR_str_(params.session).trim();
  var wantDay = NR_str_(params.day).trim();
  var out = [];

  rows.forEach(function (r) {
    var matches = wantDay
      ? String(r[2]).trim() === wantDay
      : String(r[1]).trim() === wantSession;
    if (!matches) return;
    out.push({
      id: r[12], askedAt: r[0], session: r[1], scope: r[7],
      question: r[8], askedBy: r[9], votes: Number(r[10]) || 0,
      answered: String(r[11]).trim().toLowerCase() === 'yes',
      voters: NR_str_(r[14]).split(',').filter(function (v) { return v; })
    });
  });

  out.sort(function (a, b) { return b.votes - a.votes; });
  return NR_json_({ ok: true, questions: out });
}

/** Trainer marks a question answered from the console. */
function NR_markAnswered_(p) {
  var sh = NR_tab_('ask');
  var row = NR_findRow_(sh, 13, NR_str_(p.questionId));
  if (row < 0) return NR_json_({ ok: false, error: 'not_found' });
  sh.getRange(row, 12).setValue(p.answered === false ? 'No' : 'Yes');
  return NR_json_({ ok: true });
}

/* ------------------------------------------------------ 5. TNA/TNI, anon */

/**
 * Anonymous training needs review, filled by the counsellor themselves.
 *
 * Nothing identifying is written. The payload carries no name, email, manager,
 * device key or IP, this function reads none even if one were added later, and
 * the tab has no column for one. Rows cannot be ordered back to a person beyond
 * the vertical, role and tenure band the counsellor chose to disclose.
 */
function NR_tna_(p) {
  var sh = NR_tab_('tna');
  var rid = NR_str_(p.responseId).trim() || ('T' + new Date().getTime().toString(36).toUpperCase());

  /* A retried delivery must not double-count a response. */
  if (NR_findRow_(sh, 14, rid) > 0) {
    return NR_json_({ ok: true, duplicate: true });
  }

  sh.appendRow([
    NR_now_(),
    NR_str_(p.vertical), NR_str_(p.role), NR_str_(p.tenure),
    NR_str_(p.callflow), NR_str_(p.maingap), NR_str_(p.sales),
    NR_str_(p.product), NR_str_(p.soft),
    NR_str_(p.stuck), NR_str_(p.unused), NR_str_(p.improve),
    NR_str_(p.urgency), rid
  ]);
  return NR_json_({ ok: true, duplicate: false });
}

/* --------------------------------------------- trainer dashboard access */

/**
 * The trainer access code is kept in Script Properties, never in this file and
 * never in the published page, so it cannot leak through the site's source or
 * through this repository's history.
 *
 * To set or change it, run this once from the editor with the code you want:
 *   NR_setTrainerCode('YOUR-CODE-HERE')
 * Anyone holding the old code loses access the moment you change it.
 */
function NR_setTrainerCode(code) {
  code = String(code || '').trim();
  if (code.length < 10) throw new Error('Use a code of at least 10 characters.');
  PropertiesService.getScriptProperties().setProperty('NR_TRAINER_CODE', code);
  return 'Trainer code set.';
}

/** Length-independent comparison, so timing does not leak the code. */
function NR_codeOk_(given) {
  var want = PropertiesService.getScriptProperties().getProperty('NR_TRAINER_CODE');
  if (!want) return false;
  var a = String(given || ''), b = String(want);
  if (a.length !== b.length) return false;
  var diff = 0;
  for (var i = 0; i < a.length; i++) diff |= a.charCodeAt(i) ^ b.charCodeAt(i);
  return diff === 0;
}

/**
 * Returns every anonymous TNA response for the trainer dashboard.
 *
 * The rows hold no identity, so this exposes no one even to a holder of the
 * code; the gate exists so the floor's candid feedback is not simply public.
 */
function NR_tnaRead_(p) {
  if (!NR_codeOk_(p.code)) {
    Utilities.sleep(1500);   /* blunt the speed of guessing */
    return NR_json_({ ok: false, error: 'bad_code' });
  }

  var sh = NR_tab_('tna');
  var cols = NR_HEADERS.tna.length;
  var last = sh.getLastRow();
  if (last < 2) return NR_json_({ ok: true, headers: NR_HEADERS.tna, rows: [] });

  var values = sh.getRange(2, 1, last - 1, cols).getValues();
  var rows = values.map(function (r) {
    return r.map(function (v) {
      if (v instanceof Date) return Utilities.formatDate(v, NR_TZ, 'dd MMM yyyy, HH:mm');
      return NR_str_(v);
    });
  }).filter(function (r) {
    return r.slice(0, cols - 1).join('').trim() !== '';
  });

  return NR_json_({ ok: true, headers: NR_HEADERS.tna, rows: rows });
}

/* --------------------------------------------------------------- routers */

/**
 * Returns a ContentService response for any "nr.*" action, or null so your
 * existing OND handling runs untouched.
 */
function NR_route(body) {
  if (!body || !body.action || String(body.action).indexOf('nr.') !== 0) return null;

  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(20000);
  } catch (err) {
    return NR_json_({ ok: false, error: 'busy' });
  }

  try {
    switch (body.action) {
      case 'nr.issueCode':    return NR_issueCode_(body);
      case 'nr.attendance':   return NR_attendance_(body);
      case 'nr.feedback':     return NR_feedback_(body);
      case 'nr.pulse':        return NR_pulse_(body);
      case 'nr.ask':          return NR_ask_(body);
      case 'nr.tna':          return NR_tna_(body);
      case 'nr.tnaRead':      return NR_tnaRead_(body);
      case 'nr.upvote':       return NR_upvote_(body);
      case 'nr.markAnswered': return NR_markAnswered_(body);
      case 'nr.questions':    return NR_questions_(body);
      case 'nr.checkCode':    return NR_json_(NR_checkCode_(body.code, body.session));
      default:                return NR_json_({ ok: false, error: 'unknown_action' });
    }
  } catch (err) {
    return NR_json_({ ok: false, error: String(err) });
  } finally {
    lock.releaseLock();
  }
}

/** Read-only GET routing, for the questions list and code checks. */
function NR_routeGet(params) {
  if (!params || !params.action || String(params.action).indexOf('nr.') !== 0) return null;
  try {
    switch (params.action) {
      case 'nr.questions': return NR_questions_(params);
      case 'nr.checkCode': return NR_json_(NR_checkCode_(params.code, params.session));
      default:             return NR_json_({ ok: false, error: 'unknown_action' });
    }
  } catch (err) {
    return NR_json_({ ok: false, error: String(err) });
  }
}

/**
 * Standalone fallbacks. If you would rather deploy this as its own web app than
 * merge it into the OND one, rename these two to doPost / doGet and deploy:
 * Deploy > New deployment > Web app > Execute as Me > Access: Anyone.
 */
function doPost(e) {
  var body = {};
  try { body = JSON.parse(e.postData.contents); } catch (err) {}
  return NR_route(body) || NR_json_({ ok: false, error: 'unknown_action' });
}

function doGet(e) {
  return NR_routeGet(e.parameter) || NR_json_({ ok: false, error: 'unknown_action' });
}
